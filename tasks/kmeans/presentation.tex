\section{Recap: K-Means Algorithm}
\begin{frame}{Recap: K-Means Algorithm}

\begin{enumerate}
  \item \textbf{Initialization:} Select $k$ initial centroids (randomly).
  \item \textbf{Assignment:} Assign each data point to the nearest centroid.
  \item \textbf{Update:} Recalculate centroids as the mean of assigned points.
  \item \textbf{Repeat:} Until convergence or max iterations.
\end{enumerate}

\vspace{0.4cm}
\textbf{Pseudocode:}

\begin{beamercolorbox}[rounded=true,shadow=true,sep=8pt]{block body}
\ttfamily\small
Initialize centroids = random(k)\\
While not converged:\\
\hspace*{1em}For point in data:\\
\hspace*{2em}closest = argmin(distance(point, c))\\
\hspace*{2em}assign(point, closest)\\
\hspace*{1em}For c in centroids:\\
\hspace*{2em}c = mean(assigned\_points)
\end{beamercolorbox}

\end{frame}

\section{Mini Batch}
\begin{frame}{First Optimization: OpenMp Mini Batch}
Instead of using the full dataset in every iteration, we use a
\textbf{Mini Batch K-Means} for large datasets.

\vspace{0.4cm}

\begin{itemize}
  \item The dataset is \textbf{distributed across multiple GPUs}.
  \item Each iteration processes only a \textbf{random subset} per GPU.
  \item A new random offset is chosen per iteration to sample different points.
  \item GPUs compute \textbf{partial sums and counts} for the centroids.
  \item Final centroids are obtained by \textbf{averaging across all GPUs}.
\end{itemize}

\vspace{0.4cm}

\begin{block}{Key idea}
By using different mini batches across the iterations, it reduces computation cost and memory
traffic while still converging to meaningful centeroids.
\end{block}

\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Mini Batch (Sampling)}

\begin{beamercolorbox}[rounded=true,shadow=true,sep=8pt]{block body}
\ttfamily\small
for each iteration do\\
\hspace*{1em}for each GPU do\\
\hspace*{2em}choose random offset in local data\\
\hspace*{2em}process a fixed-size batch starting at this offset\\
\hspace*{1em}end for\\
end for
\end{beamercolorbox}

\vspace{0.35cm}
Each iteration processes only a small, randomly selected subset of the
local data on every GPU.
\end{frame}


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Mini Batch (Update)}

\begin{beamercolorbox}[rounded=true,shadow=true,sep=8pt]{block body}
\ttfamily\small
for each batch do\\
\hspace*{1em}assign points to nearest centroids\\
\hspace*{1em}compute partial sums and counts\\
end for\\
aggregate results across GPUs\\
update centroids using averaged batch statistics
\end{beamercolorbox}

\vspace{0.35cm}

The centroids are updated using batch statistics.
Different mini-batches are observed over multiple iterations.

\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{What is CUDA?}
\begin{itemize}
  \item \textbf{CUDA} (Compute Unified Device Architecture) is made by NVIDIA and its general purpose is GPU computing.
  \item It enables the use of \textbf{massively parallel kernels}
        on thousands of GPU threads.
  \item CUDA offers \textbf{explicit control over memory and execution},
        including device memory, pinned memory and streams.
  \item This makes CUDA well suited for \textbf{performance critical
        numerical workloads}.
\end{itemize}

\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Why CUDA?}

\begin{itemize}
  \item \textbf{Explicit control over memory and execution}
  \begin{itemize}
    \item Pinned host memory, streams, asynchronous data transfers
  \end{itemize}

  \vspace{0.2cm}

  \item \textbf{Fine-grained kernel optimization}
  \begin{itemize}
    \item Atomic operations, shared memory, thread-level parallelism
  \end{itemize}

  \vspace{0.2cm}

  \item \textbf{Efficient multi-GPU execution}
  \begin{itemize}
    \item OpenMP orchestration with \texttt{cudaSetDevice}
  \end{itemize}
\end{itemize}

\end{frame}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\newpage
\begin{frame}[fragile]{Memory Management}

\textbf{Goal:} To reduce the data transfer and to avoid overhead and overlap communication with computation.

\vspace{0.4cm}

\begin{beamercolorbox}[rounded=true,shadow=true,sep=8pt]{block body}
{\normalfont\ttfamily\footnotesize\upshape\mdseries
\begin{lstlisting}
// Host Pinned Memory Allocation
double *pinned_x, *pinned_y;
cudaMallocHost(&pinned_x, n * sizeof(double));
cudaMallocHost(&pinned_y, n * sizeof(double));

// Copying to Device (Async)
cudaMemcpyAsync(devices[d].points_x, pinned_x + start,
               devices[d].point_count * sizeof(double),
               cudaMemcpyHostToDevice, devices[d].stream);
\end{lstlisting}
}
\end{beamercolorbox}

\end{frame}




\begin{frame}{Memory Management}

\begin{itemize}
  \item \textbf{Pinned host memory}
  \begin{itemize}
    \item Avoids paging and enables direct memory access (DMA).
    \item Results in higher and more stable host-to-device bandwidth.
  \end{itemize}

  \vspace{0.3cm}

  \item \textbf{CUDA streams}
  \begin{itemize}
    \item Allow asynchronous memory copies and kernel execution.
    \item Enable overlap of data transfers and computation.
  \end{itemize}

  \vspace{0.3cm}

  \item \textbf{Overall effect}
  \begin{itemize}
    \item Reduced idle time on the GPU.
    \item Better utilization of available hardware resources.
  \end{itemize}
\end{itemize}

\end{frame}


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{The Kernel: Nearest Centroid}

\vspace{0.2cm}
\textbf{Kernel responsibilities:}

\vspace{0.25cm}
\begin{itemize}
  \item \textbf{Thread mapping:} one thread per data point
  \item \textbf{Computation:} nearest centroid search via distance evaluation
  \item \textbf{Accumulation:} global centroid updates using \textbf{atomicAdd}
\end{itemize}

\vspace{0.25cm}
Centroid updates are shared across threads and therefore need an atomic operation.
\end{frame}

\begin{frame}[fragile]{The Kernel: Nearest Centroid}

\vspace{0.2cm}
\begin{beamercolorbox}[rounded=true,shadow=true,sep=8pt]{block body}
{\normalfont\ttfamily\footnotesize\upshape\mdseries
\begin{lstlisting}
__global__ void determine_nearest_centroid(...) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < batch_size) {
        // ... find nearest centroid logic [omitted] ...

        // Accumulate results directly in global memory
        atomicAdd(&sum_x[assignment], px);
        atomicAdd(&sum_y[assignment], py);
        atomicAdd(&count[assignment], 1);
    }
}
\end{lstlisting}
}
\end{beamercolorbox}
\vspace{0.25cm}
The performance depends on the distribution of the points across the centroids.
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Multi-GPU Strategy}

\vspace{0.15cm}
\textbf{Host parallelism via OpenMP, one control thread per GPU.}

\vspace{0.25cm}

\begin{itemize}
  \item \textbf{Mapping} OpenMP threads are mapped one to one to GPUs
  \item \textbf{Device binding:} each thread operates on a fixed CUDA device
  \item \textbf{Asynchrony:} per-GPU streams enable concurrent execution
  \item \textbf{Work split:} workload is distributed across all devices
\end{itemize}

\end{frame}

\begin{frame}[fragile]{Multi-GPU Strategy}

\vspace{0.10cm}
\begin{beamercolorbox}[rounded=true,shadow=true,sep=8pt]{block body}

\begin{lstlisting}
int num_devices = 4;
omp_set_num_threads(num_devices);

#pragma omp parallel num_threads(num_devices)
{
    int d = omp_get_thread_num();
    cudaSetDevice(d); // Bind thread to GPU

    // Create per-device stream
    cudaStreamCreate(&devices[d].stream);

    // ... Allocate and Compute ...
}
\end{lstlisting}
}
\end{beamercolorbox}

\end{frame}
